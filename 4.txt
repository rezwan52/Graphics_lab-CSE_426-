#include <GL/glut.h>
#define MAX_HISTORY 100  // Max undo/redo steps

typedef struct {
    float rotationAngle;
    float scaleFactor;
    float translateX;
    float translateY;
} TransformState;

// Transformation variables
float rotationAngle = 0.0f;
float scaleFactor = 1.0f;
float translateX = 0.0f;
float translateY = 0.0f;

// Undo/Redo stacks
TransformState undoHistory[MAX_HISTORY];
int undoTop = 0;

TransformState redoHistory[MAX_HISTORY];
int redoTop = 0;

// ---------- Push current state to Undo ----------
void pushUndoState() {
    if (undoTop < MAX_HISTORY - 1) {
        undoTop++;
        undoHistory[undoTop] = { rotationAngle, scaleFactor, translateX, translateY };
    }
    // Clear redo stack after new action
    redoTop = 0;
}

// ---------- Display ----------
void display() {
    glClear(GL_COLOR_BUFFER_BIT);
    glLoadIdentity();

    glTranslatef(translateX, translateY, 0.0f);
    glTranslatef(225, 175, 0.0f);
    glRotatef(rotationAngle, 0.0f, 0.0f, 1.0f);
    glScalef(scaleFactor, scaleFactor, 1.0f);
    glTranslatef(-225, -175, 0.0f);

    glColor3f(1.0f, 0.0f, 1.0f);
    glBegin(GL_TRIANGLES);
    glVertex2f(150.0f, 150.0f);
    glVertex2f(300.0f, 150.0f);
    glVertex2f(225.0f, 225.0f);
    glEnd();

    glFlush();
}

// ---------- Menu ----------
void menu(int option) {
    switch (option) {
        case 1: // Reset
            rotationAngle = 0.0f;
            scaleFactor = 1.0f;
            translateX = 0.0f;
            translateY = 0.0f;
            undoTop = redoTop = 0;
            undoHistory[0] = {rotationAngle, scaleFactor, translateX, translateY};
            break;

        case 2: // Rotate +15
            pushUndoState();
            rotationAngle += 15.0f;
            break;

        case 3: // Scale x1.5
            pushUndoState();
            scaleFactor *= 1.5f;
            break;

        case 4: // Translate (100,100)
            pushUndoState();
            translateX += 100.0f;
            translateY += 100.0f;
            break;

        case 5: // Scale x3
            pushUndoState();
            scaleFactor *= 3.0f;
            break;

        case 6: // Undo
            if (undoTop > 0) {
                // Save current state to redo stack
                if (redoTop < MAX_HISTORY - 1) {
                    redoTop++;
                    redoHistory[redoTop] = { rotationAngle, scaleFactor, translateX, translateY };
                }
                // Restore previous state
                undoTop--;
                rotationAngle = undoHistory[undoTop].rotationAngle;
                scaleFactor = undoHistory[undoTop].scaleFactor;
                translateX = undoHistory[undoTop].translateX;
                translateY = undoHistory[undoTop].translateY;
            }
            break;

        case 7: // Redo
            if (redoTop > 0) {
                // Save current state to undo stack
                if (undoTop < MAX_HISTORY - 1) {
                    undoTop++;
                    undoHistory[undoTop] = { rotationAngle, scaleFactor, translateX, translateY };
                }
                // Restore redo state
                rotationAngle = redoHistory[redoTop].rotationAngle;
                scaleFactor = redoHistory[redoTop].scaleFactor;
                translateX = redoHistory[redoTop].translateX;
                translateY = redoHistory[redoTop].translateY;
                redoTop--;
            }
            break;
    }

    glutPostRedisplay();
}

// ---------- Init ----------
void init() {
    glClearColor(0.0, 0.0, 0.0, 1.0);
    glMatrixMode(GL_PROJECTION);
    glLoadIdentity();
    gluOrtho2D(0.0, 800.0, 0.0, 800.0);
    glMatrixMode(GL_MODELVIEW);
}

// ---------- Main ----------
int main(int argc, char** argv) {
    glutInit(&argc, argv);
    glutInitDisplayMode(GLUT_SINGLE | GLUT_RGB);
    glutInitWindowSize(800, 800);
    glutCreateWindow("Transformations Demo (Undo/Redo)");

    init();

    // Initialize undo history
    undoHistory[0] = {rotationAngle, scaleFactor, translateX, translateY};
    undoTop = redoTop = 0;

    glutDisplayFunc(display);

    glutCreateMenu(menu);
    glutAddMenuEntry("Reset", 1);
    glutAddMenuEntry("Rotate +15", 2);
    glutAddMenuEntry("Scale x1.5", 3);
    glutAddMenuEntry("Translate (100,100)", 4);
    glutAddMenuEntry("Scale x3", 5);
    glutAddMenuEntry("Undo", 6);
    glutAddMenuEntry("Redo", 7);
    glutAttachMenu(GLUT_RIGHT_BUTTON);

    glutMainLoop();
    return 0;
}
